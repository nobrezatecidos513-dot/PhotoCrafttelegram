import os
import logging
from io import BytesIO
from urllib.parse import quote, unquote

from PIL import Image, ImageEnhance, ImageFilter, ImageOps
from aiogram import Bot, Dispatcher, executor, types
from aiogram.dispatcher.filters import Command
from aiogram.types import InlineKeyboardButton, InlineKeyboardMarkup
from dotenv import load_dotenv

logging.basicConfig(level=logging.INFO)
load_dotenv()

BOT_TOKEN = os.getenv("BOT_TOKEN")

if not BOT_TOKEN:
    raise ValueError("BOT_TOKEN não foi encontrado. Crie um arquivo .env com o seu token do Telegram.")

bot = Bot(token=BOT_TOKEN)
dp = Dispatcher(bot)

FILTERS = {
    "bw": "Preto & Branco",
    "sepia": "Sépia",
    "contrast": "Contraste",
    "vintage": "Vintage",
    "warm": "Warm",
    "sharp": "Sharp",
}


def sepia_tone(img: Image.Image) -> Image.Image:
    output = img.convert("RGB")
    pixels = output.load()
    width, height = output.size

    for y in range(height):
        for x in range(width):
            r, g, b = output.getpixel((x, y))
            tr = int(r * 0.94 + g * 0.77 + b * 0.35)
            tg = int(r * 0.76 + g * 0.68 + b * 0.28)
            tb = int(r * 0.56 + g * 0.48 + b * 0.22)
            pixels[x, y] = (min(255, tr), min(255, tg), min(255, tb))
    return output


def warm_tone(img: Image.Image) -> Image.Image:
    output = img.convert("RGB")
    pixels = output.load()
    width, height = output.size

    for y in range(height):
        for x in range(width):
            r, g, b = output.getpixel((x, y))
            pixels[x, y] = (min(255, r + 35), g, max(0, b - 15))
    return output


def vintage_tone(img: Image.Image) -> Image.Image:
    grayscale = ImageOps.grayscale(img)
    grayscale = ImageEnhance.Contrast(grayscale).enhance(1.9)
    sepia = Image.new("RGB", grayscale.size, (120, 90, 60))
    return Image.blend(grayscale.convert("RGB"), sepia, 0.28)


def apply_filter(img: Image.Image, filter_name: str) -> Image.Image:
    img = img.convert("RGB")

    if filter_name == "bw":
        return ImageOps.grayscale(img)

    if filter_name == "sepia":
        return sepia_tone(img)

    if filter_name == "contrast":
        return ImageEnhance.Contrast(img).enhance(2.2)

    if filter_name == "vintage":
        return vintage_tone(img)

    if filter_name == "warm":
        return warm_tone(img)

    if filter_name == "sharp":
        return img.filter(ImageFilter.SHARPEN)

    return img


def build_filter_keyboard(file_id: str) -> InlineKeyboardMarkup:
    keyboard = InlineKeyboardMarkup(row_width=2)
    for key, label in FILTERS.items():
        callback_data = f"filter:{quote(file_id)}:{key}"
        keyboard.insert(InlineKeyboardButton(label, callback_data=callback_data))
    return keyboard


@dp.message_handler(Command("start"))
async def cmd_start(message: types.Message):
    await message.answer(
        "Bem-vindo ao PhotoCraft Telegram!\n\n"
        "Envie uma foto e escolha um filtro para aplicar.\n\n"
        "Comandos disponíveis:\n"
        "/start - iniciar\n"
        "/help - ajuda\n"
        "/menu - abrir filtros"
    )


@dp.message_handler(Command("help"))
async def cmd_help(message: types.Message):
    await message.answer(
        "Como usar:\n"
        "1. Envie uma foto para o bot\n"
        "2. Escolha um filtro\n"
        "3. Receba a imagem editada no chat\n\n"
        "Você também pode usar /menu para abrir a seleção de filtros."
    )


@dp.message_handler(Command("menu"))
async def cmd_menu(message: types.Message):
    keyboard = InlineKeyboardMarkup(row_width=2)
    for key, label in FILTERS.items():
        keyboard.insert(InlineKeyboardButton(label, callback_data=f"preview:{key}"))

    await message.answer("Escolha um filtro:", reply_markup=keyboard)


@dp.message_handler(content_types=['photo'])
async def handle_photo(message: types.Message):
    photo = message.photo[-1]
    await message.answer_photo(
        photo.file_id,
        caption="Escolha um efeito para sua imagem:",
        reply_markup=build_filter_keyboard(photo.file_id),
    )


@dp.callback_query_handler(lambda c: c.data.startswith("preview:"))
async def handle_preview_callback(callback_query: types.CallbackQuery):
    _, filter_name = callback_query.data.split(":", 1)
    await bot.answer_callback_query(
        callback_query.id,
        text=f"Filtro selecionado: {FILTERS[filter_name]}",
        show_alert=False,
    )


@dp.callback_query_handler(lambda c: c.data.startswith("filter:"))
async def handle_filter_callback(callback_query: types.CallbackQuery):
    _, encoded_file_id, filter_name = callback_query.data.split(":", 2)
    file_id = unquote(encoded_file_id)

    await bot.answer_callback_query(
        callback_query.id,
        text=f"Aplicando filtro: {FILTERS[filter_name]}",
        show_alert=False,
    )

    try:
        file = await bot.get_file(file_id)
        file_bytes = await bot.download_file(file.file_path)
        img = Image.open(BytesIO(file_bytes)).convert("RGB")
        processed = apply_filter(img, filter_name)

        output = BytesIO()
        processed.save(output, format="JPEG", quality=90)
        output.seek(0)

        await callback_query.message.reply_photo(
            types.InputFile(output, filename=f"photo_craft_{filter_name}.jpg"),
            caption=f"✨ Efeito aplicado: {FILTERS[filter_name]}",
        )
    except Exception as e:
        logging.exception("Erro ao processar imagem")
        await callback_query.message.reply(
            "Não foi possível processar a imagem. Tente novamente com outra foto."
        )


if __name__ == "__main__":
    executor.start_polling(dp, skip_updates=True)
